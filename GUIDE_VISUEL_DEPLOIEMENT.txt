┏━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┓
┃  ÉTAPE 1 : MONGODB ATLAS (Base de données)                    [10 min]  ┃
┗━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┛

   1.1 Créer un compte
       → https://www.mongodb.com/cloud/atlas/register
       → Sign up avec ton email
       
   1.2 Créer un cluster
       → "Build a Database"
       → Plan: M0 (FREE) - 512 MB
       → Provider: AWS
       → Region: Frankfurt (eu-central-1)
       → Nom: memories-cluster
       
   1.3 Créer un utilisateur
       → Username: memoryuser
       → Password: Jojo4589!MongoDB (ou autre fort)
       ⚠️ COPIER LE MOT DE PASSE !
       
   1.4 Configurer le réseau
       → Network Access → Add IP Address
       → "Allow Access from Anywhere" (0.0.0.0/0)
       
   1.5 Récupérer la connection string
       → Connect → Connect your application
       → Driver: Node.js 5.5+
       → Copier l'URL:
         mongodb+srv://memoryuser:<password>@cluster.mongodb.net/...
       → Remplacer <password> par ton vrai mot de passe
       → Ajouter /memories après .mongodb.net
       
       EXEMPLE FINAL:
       mongodb+srv://memoryuser:Jojo4589!MongoDB@memories-cluster.
       xxxxx.mongodb.net/memories?retryWrites=true&w=majority
       
       ⚠️ SAUVEGARDER CETTE URL !

┏━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┓
┃  ÉTAPE 2 : BACKEND RENDER (API Node.js)                       [10 min]  ┃
┗━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┛

   2.1 Créer le service
       → https://dashboard.render.com
       → Sign up / Login avec GitHub
       → "New +" → "Web Service"
       → Connecter repo: jovany45/memories
       
   2.2 Configuration
       ┌─────────────────────────────────────────────────────┐
       │ Name            : memories-backend                  │
       │ Region          : Frankfurt (EU Central)            │
       │ Branch          : main                              │
       │ Root Directory  : backend                           │
       │ Runtime         : Node                              │
       │ Build Command   : npm install                       │
       │ Start Command   : npm start                         │
       │ Plan            : Free                              │
       └─────────────────────────────────────────────────────┘
       
   2.3 Générer JWT Secret
       → Ouvrir PowerShell dans ton projet
       → Exécuter: node backend/generateSecret.js
       → Copier la clé générée (64 caractères hexadécimal)
       
   2.4 Variables d'environnement (Advanced)
       ┌──────────────────┬──────────────────────────────────────────┐
       │ NODE_ENV         │ production                               │
       │ MONGODB_URI      │ (connection string MongoDB complète)     │
       │ JWT_SECRET       │ (clé générée avec generateSecret.js)     │
       │ FRONTEND_URL     │ https://memories-frontend.onrender.com   │
       └──────────────────┴──────────────────────────────────────────┘
       
   2.5 Déployer
       → "Create Web Service"
       → Attendre 5-10 minutes
       → Surveiller les logs
       → Voir "Server started on port..." = ✅ Succès
       
   2.6 Tester
       → Navigateur: https://TON-BACKEND.onrender.com/api/health
       → Réponse attendue: {"status": "OK", "message": "🚀 ..."}

┃  ÉTAPE 3 : FRONTEND RENDER (Site React)                        [5 min]  ┃
┗━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┛

   3.1 Créer le service
       → Dashboard Render → "New +" → "Static Site"
       → Même repo: jovany45/memories
       
   3.2 Configuration
       ┌─────────────────────────────────────────────────────┐
       │ Name              : memories-frontend               │
       │ Branch            : main                            │
       │ Root Directory    : frontend                        │
       │ Build Command     : npm install && npm run build    │
       │ Publish Directory : dist                            │

       │ VITE_API_URL     │ https://memories-backend.onrender.com/api│
       └──────────────────┴──────────────────────────────────────────┘
       
       ⚠️ REMPLACER "memories-backend" par le vrai nom de ton service !
       ⚠️ NE PAS OUBLIER /api à la fin !

┌──────────────────────────────────────────────────────────────────────────────┐
│                        🐛 TROUBLESHOOTING                                    │
└──────────────────────────────────────────────────────────────────────────────┘

   PROBLÈME: Backend ne démarre pas
   ─────────────────────────────────────────────────────────────────────
   ✓ Vérifier MONGODB_URI (mot de passe correct ?)
   ✓ MongoDB Atlas autorise 0.0.0.0/0 ?
   ✓ Start Command = "npm start" ?
   ✓ Voir logs Render pour erreur exacte
   
   PROBLÈME: Frontend affiche erreur réseau
   ─────────────────────────────────────────────────────────────────────
   ✓ VITE_API_URL se termine par /api ?
   ✓ Backend est en ligne ?
   ✓ FRONTEND_URL configuré dans backend ?
   ✓ Vider cache navigateur (Ctrl+Shift+R)
   
   PROBLÈME: 401 Unauthorized au login
   ─────────────────────────────────────────────────────────────────────
   ✓ JWT_SECRET configuré dans backend ?
   ✓ Re-créer compte admin: node createAdmin.js
   ✓ Clear localStorage navigateur (F12 → Application → Storage)
   
   PROBLÈME: Images ne s'affichent pas
   ─────────────────────────────────────────────────────────────────────
   ⚠️ Normal sur plan gratuit (uploads éphémères)
   ✓ Solution: Migrer vers Cloudinary
   ✓ https://cloudinary.com (plan gratuit disponible)
